            'worksheet': { label: '학습지/활동', badgeStyle: 'bg-blue-100 text-blue-700 border-blue-200' },
            'video': { label: '일반 영상', badgeStyle: 'bg-red-100 text-red-700 border-red-200' },
            'game': { label: '게임/퀴즈', badgeStyle: 'bg-purple-100 text-purple-700 border-purple-200' },
            'memo': { label: '수업 팁', badgeStyle: 'bg-yellow-100 text-yellow-700 border-yellow-200' }
        };

        const typeConfig = {
            'youtube': { icon: 'fa-youtube', color: 'text-red-500', bg: 'bg-white', label: 'Link' },
            'file': { icon: 'fa-file-lines', color: 'text-blue-500', bg: 'bg-white', label: 'File' },
            'memo': { icon: 'fa-align-left', color: 'text-yellow-600', bg: 'bg-white', label: 'Text' }
        };

        // --- 2. 인증 및 실시간 데이터 연동 ---
        async function initApp() {
            try {
                // 환경 시스템 토큰 확인을 제외하고, 선생님 프로젝트 설정에 맞춰 익명 로그인만 바로 실행합니다.
                await signInAnonymously(auth);
                
                // 인증 성공 후 데이터 로드
                loadData();
            } catch (error) {
                console.error("인증 오류:", error);
                showToast("인증에 실패했습니다. (파이어베이스 Authentication에서 '익명' 로그인을 켰는지 확인해주세요!)");
            }
        }

        function loadData() {
            const user = auth.currentUser;
            if (!user) return;

            // 공개 데이터베이스 컬렉션 연결 (RULE 1 준수)
            const colRef = collection(db, 'artifacts', appId, 'public', 'data', 'resources');
            
            onSnapshot(colRef, (snapshot) => {
                resources = [];
                snapshot.forEach(docSnap => {
                    resources.push({ id: docSnap.id, ...docSnap.data() });
                });
                
                // 최신순으로 정렬 (RULE 2 준수: JS 메모리 내 정렬)
                resources.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));
                renderAll();
            }, (error) => {
                console.error("데이터 로드 오류:", error);
                showToast("데이터를 불러오는데 오류가 발생했습니다.");
            });
        }

        // --- 3. 필터링 및 렌더링 ---
        function setGrade(grade) {
            currentGrade = grade;
            document.getElementById('currentGradeTitle').textContent = gradeMap[grade];
            
            document.querySelectorAll('.nav-grade').forEach(el => {
                el.classList.remove('bg-white/80', 'font-bold', 'text-blue-600', 'shadow-sm');
                el.classList.add('text-gray-700', 'hover:bg-white/50', 'font-medium');
            });
            const activeNav = document.getElementById('nav-' + grade);
            if(activeNav) {
                activeNav.classList.add('bg-white/80', 'font-bold', 'text-blue-600', 'shadow-sm');
                activeNav.classList.remove('text-gray-700', 'hover:bg-white/50', 'font-medium');
            }
            renderAll();
        }

        function setCategory(cat) {
            currentCategory = cat;
            
            document.querySelectorAll('.cat-pill').forEach(el => {
                el.classList.remove('bg-blue-500', 'text-white', 'border-blue-400', 'shadow-md');
                el.classList.add('glass', 'bg-white/40', 'text-gray-700', 'hover:bg-white/80');
            });
            const activeCat = document.getElementById('cat-' + cat);
            if(activeCat) {
                activeCat.classList.add('bg-blue-500', 'text-white', 'border-blue-400', 'shadow-md');
                activeCat.classList.remove('glass', 'bg-white/40', 'text-gray-700', 'hover:bg-white/80');
            }
            renderAll();
        }

        function renderAll() {
            const container = document.getElementById('dynamic-sections');
            const countLabel = document.getElementById('resultCount');
            const searchInput = document.getElementById('searchInput');
            const searchTerm = searchInput ? searchInput.value.toLowerCase() : '';
            
            const filteredRes = resources.filter(r => {
                const matchGrade = currentGrade === 'all' || r.grade === currentGrade;
                const matchCat = currentCategory === 'all' || r.category === currentCategory;
                const matchSearch = r.title.toLowerCase().includes(searchTerm) || 
                                    (r.desc && r.desc.toLowerCase().includes(searchTerm)) || 
                                    (r.content && r.content.toLowerCase().includes(searchTerm));
                return matchGrade && matchCat && matchSearch;
            });

            countLabel.textContent = `총 ${filteredRes.length}개`;
            container.innerHTML = '';

            if(filteredRes.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-20 glass rounded-3xl mt-4">
                        <div class="w-16 h-16 bg-white/50 rounded-full flex items-center justify-center mx-auto mb-4 text-gray-400 text-2xl">
                            <i class="fa-solid fa-magnifying-glass-minus"></i>
                        </div>
                        <h3 class="text-lg font-bold text-gray-800 mb-1">조건에 맞는 자료가 없습니다</h3>
                        <p class="text-sm text-gray-500">다른 필터를 선택하거나 새 자료를 추가해보세요.</p>
                    </div>`;
                return;
            }

            // Firebase 고유 ID (문자열)를 따옴표로 감싸서 전달
            const listHtml = filteredRes.map(res => {
                const tConf = typeConfig[res.uploadType] || typeConfig['file'];
                const gradeStr = gradeMap[res.grade];
                const catConf = categoryMap[res.category] || categoryMap['memo'];

                return `
                    <div onclick="openDetail('${res.id}')" class="glass-card rounded-2xl p-4 sm:p-5 flex items-center justify-between group cursor-pointer hover:bg-white/70 transition-all duration-200 relative shadow-sm hover:shadow-md">
                        <div class="flex items-start sm:items-center gap-4 w-full pr-12">
                            <div class="hidden sm:flex w-12 h-12 rounded-full ${tConf.bg} ${tConf.color} items-center justify-center flex-shrink-0 text-xl shadow-sm border border-gray-100">
                                <i class="fa-solid ${tConf.icon}"></i>
                            </div>
                            
                            <div class="flex-grow overflow-hidden">
                                <div class="flex flex-wrap gap-1.5 mb-2">
                                    <span class="text-[10px] sm:text-xs font-bold px-2 py-0.5 rounded border bg-white/60 text-gray-600 border-gray-200 shadow-sm">${gradeStr}</span>
                                    <span class="text-[10px] sm:text-xs font-bold px-2 py-0.5 rounded border ${catConf.badgeStyle} shadow-sm">${catConf.label}</span>
                                </div>
                                <h3 class="text-base sm:text-lg font-bold text-gray-900 truncate mb-1 group-hover:text-blue-600 transition-colors">${res.title}</h3>
                                <p class="text-xs sm:text-sm text-gray-500 truncate">${res.desc}</p>
                            </div>
                        </div>
                        
                        <div class="absolute right-4 flex items-center gap-2">
                            <button class="admin-btn ${isAdmin ? 'flex' : 'hidden'} w-9 h-9 rounded-full bg-red-50 text-red-500 hover:bg-red-500 hover:text-white transition items-center justify-center z-10 border border-red-100 shadow-sm" onclick="deleteResource(event, '${res.id}')" title="삭제">
                                <i class="fa-solid fa-trash-can text-sm"></i>
                            </button>
                            <div class="w-8 h-8 flex items-center justify-center rounded-full bg-gray-100/50 text-gray-400 group-hover:bg-blue-50 group-hover:text-blue-500 transition">
                                <i class="fa-solid fa-chevron-right text-sm"></i>
                            </div>
                        </div>
                    </div>
                `;
            }).join('');

            container.innerHTML = listHtml;
        }

        // --- 4. 모달 및 폼 로직 ---
        function toggleModal(modalId) {
            const modal = document.getElementById(modalId);
            const contentId = modalId === 'addModal' ? 'modalContent' : 'detailModalContent';
            const modalContent = document.getElementById(contentId);
            
            if (modal.classList.contains('hidden')) {
                modal.classList.remove('hidden');
                setTimeout(() => {
                    modalContent.classList.remove('scale-95', 'opacity-0');
                    modalContent.classList.add('scale-100', 'opacity-100');
                }, 10);
            } else {
                modalContent.classList.remove('scale-100', 'opacity-100');
                modalContent.classList.add('scale-95', 'opacity-0');
                setTimeout(() => {
                    modal.classList.add('hidden');
                }, 300);
            }
        }

        function openDetail(id) {
            const res = resources.find(r => r.id === id);
            if(!res) return;
            
            const contentArea = document.getElementById('detailModalContent');
            const gradeStr = gradeMap[res.grade];
            const catConf = categoryMap[res.category] || categoryMap['memo'];
            
            let actionHtml = '';
            if(res.uploadType === 'youtube' && res.link !== '#') {
                actionHtml = `<a href="${res.link}" target="_blank" class="w-full py-4 rounded-xl glass-btn-primary font-bold text-lg flex items-center justify-center gap-2 mt-6 transition-transform hover:-translate-y-1"><i class="fa-solid fa-play"></i> 영상 보러가기</a>`;
            } else if(res.uploadType === 'file') {
                actionHtml = `<button onclick="showToast('첨부파일 다운로드 기능은 준비 중입니다.')" class="w-full py-4 rounded-xl glass-btn-primary font-bold text-lg flex items-center justify-center gap-2 mt-6 transition-transform hover:-translate-y-1"><i class="fa-solid fa-download"></i> 다운로드 하기</button>`;
            }
            
            let bodyHtml = '';
            if(res.uploadType === 'memo' || res.content) {
                bodyHtml = `
                    <div class="mt-5">
                        <h4 class="text-xs font-bold text-gray-400 mb-2 uppercase tracking-wider">상세 내용 / 스크립트</h4>
                        <div class="p-5 bg-white/70 rounded-2xl text-gray-800 whitespace-pre-wrap leading-relaxed text-sm sm:text-base border border-gray-200/50 shadow-inner custom-scrollbar overflow-auto max-h-64">${res.content || res.desc}</div>
                    </div>`;
            }
            
            contentArea.innerHTML = `
                <div class="p-6 sm:p-8 overflow-y-auto flex-grow custom-scrollbar">
                    <div class="flex justify-between items-start mb-4">
                        <div class="flex flex-wrap gap-2 mb-2">
                            <span class="text-xs font-bold px-2 py-1 rounded bg-gray-100 text-gray-600 border border-gray-200">${gradeStr}</span>
                            <span class="text-xs font-bold px-2 py-1 rounded ${catConf.badgeStyle}">${catConf.label}</span>
                        </div>
                        <button onclick="toggleModal('detailModal')" class="text-gray-400 hover:text-gray-800 w-8 h-8 rounded-full bg-white/50 hover:bg-white border border-white/60 transition shadow-sm flex items-center justify-center">
                            <i class="fa-solid fa-times"></i>
                        </button>
                    </div>
                    
                    <h3 class="text-2xl font-bold text-gray-900 leading-tight mb-4">${res.title}</h3>
                    
                    <div class="text-gray-600 bg-white/50 p-4 rounded-xl border border-white/60 text-sm leading-relaxed shadow-sm">
                        <i class="fa-solid fa-circle-info mr-1 text-gray-400"></i> ${res.desc || '등록된 설명이 없습니다.'}
                    </div>
                    
                    ${bodyHtml}
                    ${actionHtml}
                </div>
            `;
            
            toggleModal('detailModal');
        }

        // Firebase 데이터 삭제 로직 (커스텀 확인 모달 적용)
        function deleteResource(e, id) {
            e.stopPropagation();
            
            // 브라우저 기본 confirm 대신 커스텀 모달 생성
            const confirmBox = document.createElement('div');
            confirmBox.className = 'fixed inset-0 z-[200] flex items-center justify-center p-4 bg-gray-900/40 backdrop-blur-sm transition-opacity';
            confirmBox.innerHTML = `
                <div class="glass rounded-3xl w-full max-w-sm p-6 transform transition-all shadow-2xl flex flex-col bg-white/90 border border-white/60">
                    <div class="w-12 h-12 bg-red-100 text-red-500 rounded-full flex items-center justify-center mb-4 text-xl mx-auto shadow-inner">
                        <i class="fa-solid fa-triangle-exclamation"></i>
                    </div>
                    <h3 class="text-xl font-bold text-gray-900 mb-2 text-center">자료 삭제</h3>
                    <p class="text-gray-600 mb-6 text-sm text-center">이 자료를 정말 삭제하시겠습니까?<br>이 작업은 되돌릴 수 없습니다.</p>
                    <div class="flex gap-3 justify-center w-full">
                        <button id="cancelDelBtn" class="flex-1 py-3 rounded-xl bg-gray-100 text-gray-700 font-bold hover:bg-gray-200 transition">취소</button>
                        <button id="confirmDelBtn" class="flex-1 py-3 rounded-xl bg-red-500 text-white font-bold hover:bg-red-600 transition shadow-md">삭제하기</button>
                    </div>
                </div>
            `;
            document.body.appendChild(confirmBox);

            document.getElementById('cancelDelBtn').onclick = () => confirmBox.remove();
            document.getElementById('confirmDelBtn').onclick = async () => {
                confirmBox.remove();
                try {
                    await deleteDoc(doc(db, 'artifacts', appId, 'public', 'data', 'resources', id));
                    showToast('자료가 성공적으로 삭제되었습니다.');
                } catch(err) {
                    console.error("삭제 오류:", err);
                    showToast('삭제 중 오류가 발생했습니다.');
                }
            };
        }

        function toggleInputType(type) {
            const urlGroup = document.getElementById('urlInputGroup');
            const fileGroup = document.getElementById('fileInputGroup');
            const memoGroup = document.getElementById('memoInputGroup');
            
            [urlGroup, fileGroup, memoGroup].forEach(el => el.classList.add('hidden'));
            document.getElementById('urlInput').removeAttribute('required');
            document.getElementById('memoInput').removeAttribute('required');

            ['labelYoutube', 'labelFile', 'labelMemo'].forEach(id => {
                document.getElementById(id).classList.remove('border-blue-400', 'bg-blue-50/30');
                document.getElementById(id).classList.add('border-transparent');
            });

            if (type === 'youtube') {
                urlGroup.classList.remove('hidden');
                document.getElementById('urlInput').setAttribute('required', 'true');
                document.getElementById('labelYoutube').classList.add('border-blue-400', 'bg-blue-50/30');
            } else if (type === 'file') {
                fileGroup.classList.remove('hidden');
                document.getElementById('labelFile').classList.add('border-blue-400', 'bg-blue-50/30');
            } else if (type === 'memo') {
                memoGroup.classList.remove('hidden');
                document.getElementById('memoInput').setAttribute('required', 'true');
                document.getElementById('labelMemo').classList.add('border-blue-400', 'bg-blue-50/30');
            }
        }

        // Firebase 데이터 추가 로직
        async function submitForm(e) {
            e.preventDefault();
            
            const grade = document.getElementById('gradeSelect').value;
            const category = document.getElementById('categorySelect').value;
            const uploadType = document.querySelector('input[name="uploadType"]:checked').value;
            
            const title = document.getElementById('titleInput').value;
            const desc = document.getElementById('descInput').value;
            const link = document.getElementById('urlInput')?.value || '#';
            const content = document.getElementById('memoInput')?.value || '';
            
            const newRes = {
                grade,
                category,
                uploadType,
                title,
                desc,
                link,
                content,
                createdAt: Date.now() // 정렬을 위한 타임스탬프
            };
            
            try {
                // public 컬렉션에 새 문서 추가
                await addDoc(collection(db, 'artifacts', appId, 'public', 'data', 'resources'), newRes);
                
                toggleModal('addModal');
                document.getElementById('resourceForm').reset();
                toggleInputType('youtube'); 
                
                setGrade('all');
                setCategory('all');
                document.getElementById('searchInput').value = '';
                
                showToast('새 자료가 성공적으로 보관함에 등록되었습니다!');
            } catch (error) {
                console.error("추가 오류:", error);
                showToast('자료 업로드 중 오류가 발생했습니다.');
            }
        }

        // --- 5. 기타 유틸리티 ---
        function toggleAdminMode() {
            isAdmin = !isAdmin;
            const addBtns = document.querySelectorAll('.add-resource-btn');
            
            const icons = ['adminBtnIcon', 'mobileAdminBtnIcon'];
            const texts = ['adminBtnText', 'mobileAdminBtnText'];
            
            if (isAdmin) {
                addBtns.forEach(btn => btn.classList.remove('hidden'));
                icons.forEach(id => document.getElementById(id)?.classList.replace('fa-lock', 'fa-unlock'));
                texts.forEach(id => { if(document.getElementById(id)) document.getElementById(id).textContent = '관리자 종료'; });
                showToast('관리자 모드 활성화: 항목을 추가/삭제할 수 있습니다.');
            } else {
                addBtns.forEach(btn => btn.classList.add('hidden'));
                icons.forEach(id => document.getElementById(id)?.classList.replace('fa-unlock', 'fa-lock'));
                texts.forEach(id => { if(document.getElementById(id)) document.getElementById(id).textContent = '관리자 모드'; });
                showToast('관리자 모드가 종료되었습니다.');
            }
            renderAll();
        }

        function toggleMobileMenu() {
            document.getElementById('mobileMenu').classList.toggle('hidden');
        }

        function showToast(message) {
            const toast = document.getElementById('toast');
            document.getElementById('toastMsg').textContent = message;
            toast.classList.remove('translate-y-20', 'opacity-0');
            setTimeout(() => toast.classList.add('translate-y-20', 'opacity-0'), 3000);
        }

        window.addEventListener('scroll', () => {
            const nav = document.querySelector('nav');
            if (window.scrollY > 10) {
                nav.classList.add('bg-white/70', 'shadow-sm');
                nav.classList.remove('glass');
            } else {
                nav.classList.remove('bg-white/70', 'shadow-sm');
                nav.classList.add('glass');
            }
        });

        // --- 모듈 스코프 해결을 위한 전역(window) 객체 연결 ---
        window.setGrade = setGrade;
        window.setCategory = setCategory;
        window.renderAll = renderAll;
        window.toggleModal = toggleModal;
        window.openDetail = openDetail;
        window.deleteResource = deleteResource;
        window.toggleInputType = toggleInputType;
        window.submitForm = submitForm;
        window.toggleAdminMode = toggleAdminMode;
        window.toggleMobileMenu = toggleMobileMenu;
        window.showToast = showToast;

        // 초기 앱 구동
        initApp();
    </script>
